import React, { useState } from "react";

export function TrainingPage({ cardsets }) {
  const [selectedSet, setSelectedSet] = useState(null);
  const [currentCardIndex, setCurrentCardIndex] = useState(0);
  const [showAnswer, setShowAnswer] = useState(false);

  // Если набор не выбран, показываем список наборов
  if (!selectedSet) {
    return (
      <div>
        <h2>Режимы обучения</h2>
        <div>
          <h3>Выберите набор для тренировки:</h3>
          {cardsets.map((set) => (
            <div
              key={set.id}
              style={{
                border: "1px solid #ccc",
                padding: "10px",
                margin: "10px",
              }}
            >
              <h4>{set.title}</h4>
              <p>Карточек: {set.cards?.length || 0}</p>
              <button onClick={() => setSelectedSet(set)}>
                Начать тренировку
              </button>
            </div>
          ))}
        </div>
      </div>
    );
  }

  // Если набор выбран, показываем карточки
  const currentCard = selectedSet.cards?.[currentCardIndex];

  if (!currentCard) {
    return (
      <div>
        <h2>Тренировка завершена! 🎉</h2>
        <p>Вы прошли все карточки в наборе "{selectedSet.title}"</p>
        <button
          onClick={() => {
            setSelectedSet(null);
            setCurrentCardIndex(0);
            setShowAnswer(false);
          }}
        >
          Выбрать другой набор
        </button>
      </div>
    );
  }

  return (
    <div>
      <h2>Тренировка: {selectedSet.title}</h2>
      <p>
        Карточка {currentCardIndex + 1} из {selectedSet.cards.length}
      </p>

      <div
        style={{
          border: "2px solid #007bff",
          padding: "20px",
          margin: "20px",
          minHeight: "200px",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          cursor: "pointer",
        }}
        onClick={() => setShowAnswer(!showAnswer)}
      >
        {showAnswer ? (
          <div>
            <h3>Ответ:</h3>
            <p>{currentCard.back}</p>
          </div>
        ) : (
          <div>
            <h3>Вопрос:</h3>
            <p>{currentCard.front}</p>
          </div>
        )}
      </div>

      <div>
        {showAnswer && (
          <button
            onClick={() => {
              setShowAnswer(false);
              setCurrentCardIndex(currentCardIndex + 1);
            }}
          >
            Следующая карточка
          </button>
        )}
        <button
          onClick={() => {
            setSelectedSet(null);
            setCurrentCardIndex(0);
            setShowAnswer(false);
          }}
        >
          Завершить тренировку
        </button>
      </div>
    </div>
  );
}
