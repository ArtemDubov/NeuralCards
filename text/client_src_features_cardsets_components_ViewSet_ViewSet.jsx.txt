import React from "react";
import { useLanguage } from "../../../../contexts/LanguageContext";
import "./ViewSet.css";

const ViewSet = ({
  set,
  onBack,
  onAddCard,
  onViewCard,
  onDeleteSet,
  onDeleteCard,
}) => {
  const { t } = useLanguage();

  return (
    <div className="view-set-container">
      <div className="view-set-header">
        <button className="btn-tp3" onClick={onBack}>
          {t("sets.back")}
        </button>
        <h2>{set.title}</h2>
        <div className="header-actions">
          <button className="btn-tp1" onClick={onAddCard}>
            {t("sets.add.card")}
          </button>
          <button className="btn-tp4" onClick={() => onDeleteSet(set)}>
            {t("sets.delete")}
          </button>
        </div>
      </div>

      {set.tags && set.tags.length > 0 && (
        <div className="set-tags">
          {set.tags.split(",").map((tag, index) => (
            <span key={index} className="tag">
              {tag.trim()}
            </span>
          ))}
        </div>
      )}

      <div className="cards-grid">
        {set.cards && set.cards.length > 0 ? (
          set.cards.map((card) => (
            <div key={card.id} className="container-tp4 card-item">
              <div className="card-preview">
                {/* Front side preview */}
                <div className="card-side">
                  <h4>{t("cards.front")}</h4>
                  {card.imageUrl && (
                    <img
                      src={card.imageUrl}
                      alt="Front"
                      className="card-image"
                    />
                  )}
                  {card.audioUrl && <div className="audio-indicator">🎵</div>}
                  <p>{card.front}</p>
                </div>

                {/* Back side preview */}
                <div className="card-side">
                  <h4>{t("cards.back")}</h4>
                  {card.backImageUrl && (
                    <img
                      src={card.backImageUrl}
                      alt="Back"
                      className="card-image"
                    />
                  )}
                  {card.backAudioUrl && (
                    <div className="audio-indicator">🎵</div>
                  )}
                  <p>{card.back}</p>
                </div>
              </div>

              <div className="card-actions">
                <button className="btn-tp3" onClick={() => onViewCard(card)}>
                  {t("sets.view")}
                </button>
                <button className="btn-tp4" onClick={() => onDeleteCard(card)}>
                  {t("sets.delete")}
                </button>
              </div>
            </div>
          ))
        ) : (
          <div className="container-tp3 empty-cards">
            <p>{t("sets.cards.empty")}</p>
          </div>
        )}
      </div>
    </div>
  );
};

export default ViewSet;
