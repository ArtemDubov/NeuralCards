import React from "react";
import { useLanguage } from "../../../../contexts/LanguageContext";
import "./CardsetList.css";

const CardsetList = ({ cardsets, handleViewSet, showDeleteModal }) => {
  const { t } = useLanguage();

  return (
    <div className="cardset-list">
      {cardsets.map((set) => {
        // Безопасная обработка тегов
        const tagsArray = set.tags
          ? typeof set.tags === "string"
            ? set.tags
                .split(",")
                .map((tag) => tag.trim())
                .filter((tag) => tag)
            : Array.isArray(set.tags)
            ? set.tags
            : []
          : [];

        return (
          <div key={set.id} className="container-tp2">
            <div className="cardset-header">
              <h3>{set.title}</h3>
              <span className="cards-count">
                {set.cards ? set.cards.length : 0} {t("sets.cards_count")}
              </span>
            </div>

            {tagsArray.length > 0 && (
              <div className="cardset-tags">
                {tagsArray.map((tag, index) => (
                  <span key={index} className="tag">
                    {tag}
                  </span>
                ))}
              </div>
            )}

            <div className="cardset-actions">
              <button className="btn-tp1" onClick={() => handleViewSet(set)}>
                {t("sets.view")}
              </button>
              <button
                className="btn-tp4"
                onClick={() => showDeleteModal(set, "set")}
              >
                {t("sets.delete")}
              </button>
            </div>
          </div>
        );
      })}
    </div>
  );
};

export default CardsetList;
