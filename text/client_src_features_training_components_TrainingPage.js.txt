import React, { useState } from "react";

export function TrainingPage({ cardsets }) {
  const [selectedSet, setSelectedSet] = useState(null);
  const [currentCardIndex, setCurrentCardIndex] = useState(0);
  const [showAnswer, setShowAnswer] = useState(false);

  // Если набор не выбран, показываем список наборов
  if (!selectedSet) {
    return (
      <div className="training-container">
        <h2>Режимы обучения</h2>
        <div className="sets-selection">
          <h3>Выберите набор для тренировки:</h3>
          <div className="sets-grid">
            {cardsets.map((set) => (
              <div key={set.id} className="training-set-card">
                <h4>{set.title}</h4>
                <p className="cards-count">
                  Карточек: {set.cards?.length || 0}
                </p>
                <button
                  className="action-button"
                  onClick={() => setSelectedSet(set)}
                >
                  Начать тренировку
                </button>
              </div>
            ))}
          </div>
        </div>
      </div>
    );
  }

  // Если набор выбран, показываем карточки
  const currentCard = selectedSet.cards?.[currentCardIndex];

  if (!currentCard) {
    return (
      <div className="training-container completion-screen">
        <h2>Тренировка завершена! 🎉</h2>
        <p>Вы прошли все карточки в наборе "{selectedSet.title}"</p>
        <button
          className="action-button"
          onClick={() => {
            setSelectedSet(null);
            setCurrentCardIndex(0);
            setShowAnswer(false);
          }}
        >
          Выбрать другой набор
        </button>
      </div>
    );
  }

  return (
    <div className="training-container">
      <div className="training-header">
        <h2>Тренировка: {selectedSet.title}</h2>
        <div className="progress-info">
          Карточка {currentCardIndex + 1} из {selectedSet.cards.length}
        </div>
      </div>

      <div
        className={`training-card ${showAnswer ? "show-answer" : ""}`}
        onClick={() => setShowAnswer(!showAnswer)}
      >
        {showAnswer ? (
          <div className="card-content">
            <h3>Ответ:</h3>
            <p className="card-text">{currentCard.back}</p>

            {/* МЕДИА НА ОБОРОТНОЙ СТОРОНЕ */}
            {currentCard.backImageUrl && (
              <div className="card-media">
                <img
                  src={`http://localhost:5002${currentCard.backImageUrl}`}
                  alt=""
                  className="card-image"
                />
              </div>
            )}

            {currentCard.backAudioUrl && (
              <div className="card-media">
                <audio controls className="card-audio">
                  <source
                    src={`http://localhost:5002${currentCard.backAudioUrl}`}
                    type="audio/mpeg"
                  />
                  <source
                    src={`http://localhost:5002${currentCard.backAudioUrl}`}
                    type="audio/wav"
                  />
                  <source
                    src={`http://localhost:5002${currentCard.backAudioUrl}`}
                    type="audio/ogg"
                  />
                  Ваш браузер не поддерживает аудио элемент.
                </audio>
              </div>
            )}
          </div>
        ) : (
          <div className="card-content">
            <h3>Вопрос:</h3>
            <p className="card-text">{currentCard.front}</p>

            {/* МЕДИА НА ЛИЦЕВОЙ СТОРОНЕ */}
            {currentCard.imageUrl && (
              <div className="card-media">
                <img
                  src={`http://localhost:5002${currentCard.imageUrl}`}
                  alt=""
                  className="card-image"
                />
              </div>
            )}

            {currentCard.audioUrl && (
              <div className="card-media">
                <audio controls className="card-audio">
                  <source
                    src={`http://localhost:5002${currentCard.audioUrl}`}
                    type="audio/mpeg"
                  />
                  <source
                    src={`http://localhost:5002${currentCard.audioUrl}`}
                    type="audio/wav"
                  />
                  <source
                    src={`http://localhost:5002${currentCard.audioUrl}`}
                    type="audio/ogg"
                  />
                  Ваш браузер не поддерживает аудио элемент.
                </audio>
              </div>
            )}
          </div>
        )}
        <div className="card-hint">👆 Нажмите, чтобы перевернуть</div>
      </div>

      <div className="training-controls">
        {showAnswer && (
          <button
            className="action-button next-button"
            onClick={() => {
              setShowAnswer(false);
              setCurrentCardIndex(currentCardIndex + 1);
            }}
          >
            Следующая карточка →
          </button>
        )}
        <button
          className="secondary-button"
          onClick={() => {
            setSelectedSet(null);
            setCurrentCardIndex(0);
            setShowAnswer(false);
          }}
        >
          Завершить тренировку
        </button>
      </div>
    </div>
  );
}
