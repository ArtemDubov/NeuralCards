import React from "react";
import { useLanguage } from "../../../../contexts/LanguageContext";
import "./CardModal.css";

const CardModal = ({
  isOpen,
  onClose,
  onSubmit,
  title,
  cardFrontText,
  setCardFrontText,
  cardBackText,
  setCardBackText,
  frontImage,
  setFrontImage,
  backImage,
  setBackImage,
  frontAudio,
  setFrontAudio,
  backAudio,
  setBackAudio,
  isUploading,
  submitText,
}) => {
  const { t } = useLanguage();

  if (!isOpen) return null;

  const handleFileChange = (e, setFile) => {
    const file = e.target.files[0];
    if (file) setFile(file);
  };

  const removeFile = (setFile) => {
    setFile(null);
  };

  return (
    <div className="modal-overlay">
      <div className="container-tp1 modal-content card-modal">
        <h2>{title}</h2>

        <form onSubmit={onSubmit}>
          {/* Front Side */}
          <div className="form-section">
            <h3>{t("cards.front")}</h3>
            <textarea
              value={cardFrontText}
              onChange={(e) => setCardFrontText(e.target.value)}
              placeholder={t("cards.front")}
              rows="3"
            />

            <div className="file-upload-section">
              <label className="file-upload-btn">
                <span>{t("cards.add.image")}</span>
                <input
                  type="file"
                  accept="image/*"
                  onChange={(e) => handleFileChange(e, setFrontImage)}
                  style={{ display: "none" }}
                />
              </label>

              <label className="file-upload-btn">
                <span>{t("cards.add.audio")}</span>
                <input
                  type="file"
                  accept="audio/*"
                  onChange={(e) => handleFileChange(e, setFrontAudio)}
                  style={{ display: "none" }}
                />
              </label>
            </div>

            {(frontImage || frontAudio) && (
              <div className="file-preview">
                {frontImage && (
                  <div className="preview-item">
                    <span>📷 {frontImage.name}</span>
                    <button
                      type="button"
                      onClick={() => removeFile(setFrontImage)}
                      className="btn-tp7"
                    >
                      {t("cards.remove")}
                    </button>
                  </div>
                )}
                {frontAudio && (
                  <div className="preview-item">
                    <span>🎵 {frontAudio.name}</span>
                    <button
                      type="button"
                      onClick={() => removeFile(setFrontAudio)}
                      className="btn-tp7"
                    >
                      {t("cards.remove")}
                    </button>
                  </div>
                )}
              </div>
            )}
          </div>

          {/* Back Side */}
          <div className="form-section">
            <h3>{t("cards.back")}</h3>
            <textarea
              value={cardBackText}
              onChange={(e) => setCardBackText(e.target.value)}
              placeholder={t("cards.back")}
              rows="3"
            />

            <div className="file-upload-section">
              <label className="file-upload-btn">
                <span>{t("cards.add.image")}</span>
                <input
                  type="file"
                  accept="image/*"
                  onChange={(e) => handleFileChange(e, setBackImage)}
                  style={{ display: "none" }}
                />
              </label>

              <label className="file-upload-btn">
                <span>{t("cards.add.audio")}</span>
                <input
                  type="file"
                  accept="audio/*"
                  onChange={(e) => handleFileChange(e, setBackAudio)}
                  style={{ display: "none" }}
                />
              </label>
            </div>

            {(backImage || backAudio) && (
              <div className="file-preview">
                {backImage && (
                  <div className="preview-item">
                    <span>📷 {backImage.name}</span>
                    <button
                      type="button"
                      onClick={() => removeFile(setBackImage)}
                      className="btn-tp7"
                    >
                      {t("cards.remove")}
                    </button>
                  </div>
                )}
                {backAudio && (
                  <div className="preview-item">
                    <span>🎵 {backAudio.name}</span>
                    <button
                      type="button"
                      onClick={() => removeFile(setBackAudio)}
                      className="btn-tp7"
                    >
                      {t("cards.remove")}
                    </button>
                  </div>
                )}
              </div>
            )}
          </div>

          <div className="modal-actions">
            <button
              type="button"
              className="btn-tp3"
              onClick={onClose}
              disabled={isUploading}
            >
              {t("cards.cancel")}
            </button>
            <button type="submit" className="btn-tp1" disabled={isUploading}>
              {submitText}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};

export default CardModal;
