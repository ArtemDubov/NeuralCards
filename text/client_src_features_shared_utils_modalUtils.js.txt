// Утилиты для управления модальными окнами

/**
 * Сбрасывает состояние кликов (аналог resetModalState для обратной совместимости)
 */
export const resetClickState = () => {
  setTimeout(() => {
    if (document.activeElement) {
      document.activeElement.blur();
    }
    document.body.focus();
  }, 10);
};

/**
 * Сбрасывает фокус и восстанавливает взаимодействие после закрытия модального окна
 */
export const resetModalState = () => {
  resetClickState();
};

/**
 * Обработчик закрытия модального окна с автоматическим сбросом состояния
 */
export const createModalCloseHandler = (onClose) => {
  return (e) => {
    if (e) {
      e.stopPropagation();
      if (e.currentTarget) {
        e.currentTarget.blur();
      }
    }

    if (onClose) onClose();
    resetModalState();
  };
};

/**
 * Обработчик подтверждения в модальном окне с автоматическим сбросом состояния
 */
export const createModalConfirmHandler = (onConfirm) => {
  return (e) => {
    if (e) {
      e.stopPropagation();
      if (e.currentTarget) {
        e.currentTarget.blur();
      }
    }

    if (onConfirm) onConfirm();
    resetModalState();
  };
};
